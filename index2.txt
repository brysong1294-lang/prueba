/* ========================= API ========================= */
async function xtreamFetch(action, params={}){
  const s = state.session;
  const url = new URL(s.server.replace(/\/$/,'') + '/player_api.php');
  url.searchParams.set('username', s.username);
  url.searchParams.set('password', s.password);
  if(action) url.searchParams.set('action', action);
  Object.entries(params).forEach(([k,v])=>url.searchParams.set(k,v));
  const res = await fetch(url.toString());
  if(!res.ok) throw new Error('HTTP '+res.status);
  return res.json();
}
function buildStreamUrl(type, id, extension){
  const s = state.session; const base = s.server.replace(/\/$/,'');
  const ext = extension || 'mp4';
  if(type==='live') return `${base}/live/${s.username}/${s.password}/${id}.m3u8`;
  if(type==='movie') return `${base}/movie/${s.username}/${s.password}/${id}.${ext}`;
  return `${base}/series/${s.username}/${s.password}/${id}.${ext}`;
}
function normalizeItem(raw){
  const rawIcon = raw.stream_icon || raw.cover || raw.cover_big || '';
  const icon = (rawIcon && rawIcon !== '0') ? rawIcon : '';
  return {
    ...raw,
    id: raw.stream_id ?? raw.series_id ?? raw.vod_id,
    name: raw.name || '',
    icon: icon,
    rating: parseFloat(raw.rating || raw.rating_5based || 0) || null,
    added: raw.added ? parseInt(raw.added) : null,
    extension: raw.container_extension || 'mp4',
    category_id: raw.category_id,
  };
}

function getVisibleContent(type){
  const all = state.content[type]||[];
  if(!state.parental.enabled || state.parental.unlocked) return all;
  const lockedSet = state.adultCategoryIds[type] || new Set();
  return all.filter(i=>!lockedSet.has(String(i.category_id)));
}
function loadParentalState(){
  const saved = storageGet('xtream_parental', {enabled:false, pin:''});
  state.parental = { enabled: !!saved.enabled, pin: saved.pin||'', unlocked:false };
}
function saveParentalState(){
  storageSet('xtream_parental', {enabled: state.parental.enabled, pin: state.parental.pin});
}
function refreshVisibleContent(){
  if(state.currentTab==='live'){ renderLiveCategories(); renderChannelsList(state.currentCategory); }
  else if(state.currentTab==='movies' || state.currentTab==='series'){ renderVodCategories(); applyVodFilters(); }
}

function wireParentalActions(){
  const p = state.parental;
  const errBox = ()=>document.getElementById('parental-error');
  const showErr = (msg)=>{ const e=errBox(); if(e){ e.textContent=msg; e.classList.remove('hidden'); } };

  const saveBtn = document.getElementById('pp-save');
  if(saveBtn) saveBtn.onclick = ()=>{
    const a = document.getElementById('pp-new-pin').value.trim();
    const b = document.getElementById('pp-confirm-pin').value.trim();
    if(!/^\d{4,6}$/.test(a)){ showErr('El PIN debe tener entre 4 y 6 dígitos.'); return; }
    if(a!==b){ showErr('Los PIN no coinciden.'); return; }
    p.pin=a; p.enabled=true; p.unlocked=false; saveParentalState();
    toast('Control parental activado','success'); renderParentalModal(); refreshVisibleContent();
  };
  const enableBtn = document.getElementById('pp-enable');
  if(enableBtn) enableBtn.onclick = ()=>{ p.enabled=true; p.unlocked=false; saveParentalState(); toast('Protección activada','success'); renderParentalModal(); refreshVisibleContent(); };
  const changeBtn = document.getElementById('pp-change-pin');
  if(changeBtn) changeBtn.onclick = ()=>{
    const oldPin = prompt('Ingresa tu PIN actual:');
    if(oldPin===null) return;
    if(oldPin !== p.pin){ toast('PIN incorrecto','error'); return; }
    const newPin = prompt('Ingresa el nuevo PIN (4-6 dígitos):');
    if(!newPin) return;
    if(!/^\d{4,6}$/.test(newPin)){ toast('PIN inválido','error'); return; }
    p.pin=newPin; saveParentalState(); toast('PIN actualizado','success');
  };
  const removeBtn = document.getElementById('pp-remove-pin');
  if(removeBtn) removeBtn.onclick = ()=>{
    const oldPin = prompt('Ingresa tu PIN actual para eliminarlo:');
    if(oldPin===null) return;
    if(oldPin !== p.pin){ toast('PIN incorrecto','error'); return; }
    p.pin=''; p.enabled=false; p.unlocked=false; saveParentalState();
    toast('Control parental eliminado','info'); renderParentalModal(); refreshVisibleContent();
  };
  const unlockBtn = document.getElementById('pp-unlock');
  if(unlockBtn) unlockBtn.onclick = ()=>{
    const val = document.getElementById('pp-unlock-pin').value.trim();
    if(val !== p.pin){ showErr('PIN incorrecto.'); return; }
    p.unlocked=true; toast('Contenido desbloqueado','success'); renderParentalModal(); refreshVisibleContent();
  };
  const relockBtn = document.getElementById('pp-relock');
  if(relockBtn) relockBtn.onclick = ()=>{ p.unlocked=false; toast('Contenido bloqueado nuevamente','info'); renderParentalModal(); refreshVisibleContent(); };
  const disableBtn = document.getElementById('pp-disable');
  if(disableBtn) disableBtn.onclick = ()=>{
    const val = prompt('Ingresa tu PIN para desactivar la protección:');
    if(val===null) return;
    if(val !== p.pin){ toast('PIN incorrecto','error'); return; }
    p.enabled=false; p.unlocked=false; saveParentalState();
    toast('Protección desactivada','info'); renderParentalModal(); refreshVisibleContent();
  };
}

  document.getElementById('profile-dropdown').classList.add('hidden');
  toast('Actualizando contenido...','info');

  state.content = {};
  state.categories = {};

    if(state.currentTab==='live') await openLiveTab();
    else if(state.currentTab==='movies') await openVodTab('movie');

  }catch(e){ toast('Error al actualizar contenido','error'); }

/* ========================= DATA LOADING ========================= */
async function ensureTypeLoaded(type){
  if(state.loaded[type]) return;
  const catAction = {live:'get_live_categories', movie:'get_vod_categories', series:'get_series_categories'}[type];
  const listAction = {live:'get_live_streams', movie:'get_vod_streams', series:'get_series'}[type];
  const [cats, all] = await Promise.all([xtreamFetch(catAction), xtreamFetch(listAction)]);
  state.categories[type] = Array.isArray(cats)?cats:[];
  computeAdultCategories(type, state.categories[type]);
  const normalized = (Array.isArray(all)?all:[]).map(normalizeItem);
  state.content[type] = normalized;
  const counts = {};
  normalized.forEach(it=>{ counts[it.category_id] = (counts[it.category_id]||0)+1; });
  state.counts[type] = counts;
  state.loaded[type] = true;
}

/* ========================= RENDER: CATEGORIES ========================= */
function renderCategoriesCol(container, type, activeId, onSelect){
  const catsAll = state.categories[type]||[];
  const cats = catsAll.filter(c=>!isCategoryLocked(type, c.category_id));
  const counts = state.counts[type]||{};
  const total = getVisibleContent(type).length;
  container.innerHTML = `
    <div class="cat-search"><input placeholder="Buscar categorías..." id="cs-${type}"></div>
    <div class="cat-list" id="cl-${type}">
      <div class="cat-item ${!activeId?'active':''}" data-id="">Todos <span class="badge">${total}</span></div>
      ${cats.map(c=>`<div class="cat-item ${String(activeId)===String(c.category_id)?'active':''}" data-id="${esc(c.category_id)}">${esc(c.category_name)} <span class="badge">${counts[c.category_id]||0}</span></div>`).join('')}
    </div>`;
  container.querySelectorAll('.cat-item').forEach(el=>{
    el.onclick = ()=> onSelect(el.dataset.id || null);
  });
  container.querySelector(`#cs-${type}`).oninput = (e)=>{
    const q = e.target.value.toLowerCase();
    container.querySelectorAll('.cat-item').forEach(el=>{
      el.style.display = el.textContent.toLowerCase().includes(q) ? '' : 'none';
    });
  };
}

/* ========================= LIVE TV ========================= */
function stopLiveVideo(){
  const video = document.getElementById('live-video');
  if(!video) return;
  try{ video.pause(); }catch(e){}
  if(video._hls){ try{ video._hls.destroy(); }catch(e){} video._hls=null; }
  video.removeAttribute('src');
  try{ video.load(); }catch(e){}
  clearInterval(video._progressInterval);
  state.currentChannelId = null;
}
async function openLiveTab(){
  document.getElementById('live-current-title').textContent = 'Selecciona un canal';
  document.getElementById('live-current-sub').textContent = 'TV en vivo';
  try{ await ensureTypeLoaded('live'); }catch(e){ toast('Error cargando canales','error'); return; }
  renderLiveCategories();
  renderChannelsList(state.currentCategory);
}

function renderChannelsList(catId){
  const list = catId ? state.content.live.filter(c=>String(c.category_id)===String(catId)) : getVisibleContent('live');
  state.liveChannels = list;
  const col = document.getElementById('live-channels-col');
  col.innerHTML = `<div class="cat-search"><input placeholder="Buscar canal..." id="live-search"></div><div id="live-list"></div>`;
  const renderList = (items)=>{
    document.getElementById('live-list').innerHTML = items.map(ch=>`
      <div class="channel-item ${state.currentChannelId===ch.id?'active':''}" data-id="${ch.id}">
        <img src="${esc(ch.icon)}" onerror="this.src='logo.png'">
        <span>${esc(ch.name)}</span>
      </div>`).join('') || '<p style="color:var(--text-dim);font-size:.8rem;padding:1rem 0;">No hay canales.</p>';
    document.querySelectorAll('#live-list .channel-item').forEach(el=>{
      el.onclick = ()=>{
        const ch = list.find(c=>String(c.id)===el.dataset.id);
        if(ch) playLive(ch);
        document.querySelectorAll('#live-list .channel-item').forEach(x=>x.classList.remove('active'));
        el.classList.add('active');
        setSectionForView('view-live','player');
      };
    });
  };
  renderList(list);
  document.getElementById('live-search').oninput = (e)=>{
    const q = e.target.value.toLowerCase();
    renderList(list.filter(c=>c.name.toLowerCase().includes(q)));
  };
}
function playLive(channel){
  state.currentChannelId = channel.id;
  document.getElementById('live-current-title').textContent = channel.name;
  document.getElementById('live-current-sub').textContent = 'TV en Vivo · HLS';
  const favBtn = document.getElementById('live-fav-btn');
  const fav = isFavorite('live', channel.id);
  favBtn.textContent = fav ? '❤️ Favorito' : '🤍 Favorito';
  favBtn.onclick = ()=>{ toggleFavorite('live', channel); favBtn.textContent = isFavorite('live',channel.id)?'❤️ Favorito':'🤍 Favorito'; };

  const url = buildStreamUrl('live', channel.id, 'm3u8');
  const video = document.getElementById('live-video');
  initHlsPlayer(video, url, { isLive:true, controller: state.liveController, url });
  loadEpg(channel.id);
}
async function loadEpg(streamId){
  const box = document.getElementById('epg-content');
  box.innerHTML = '<p style="color:var(--text-dim);font-size:.8rem;">Cargando programación...</p>';
  try{
    const res = await xtreamFetch('get_short_epg', {stream_id:streamId, limit:2});
    const list = res && res.epg_listings ? res.epg_listings : [];
    if(!list.length){ box.innerHTML = '<p style="color:var(--text-dim);font-size:.8rem;">No hay información de EPG disponible.</p>'; return; }
    box.innerHTML = list.map((ep,i)=>{
      const title = decodeEpg(ep.title);
      const desc = decodeEpg(ep.description);
      const start = ep.start_timestamp ? parseInt(ep.start_timestamp)*1000 : (ep.start? new Date(ep.start.replace(' ','T')).getTime() : null);
      const stop = ep.stop_timestamp ? parseInt(ep.stop_timestamp)*1000 : (ep.end? new Date(ep.end.replace(' ','T')).getTime() : null);
      let pctHtml = '';
      if(i===0 && start && stop){
        const now = Date.now();
        const pct = Math.min(100, Math.max(0, ((now-start)/(stop-start))*100));
        pctHtml = `<div class="epg-progress"><div class="epg-progress-bar" style="width:${pct}%"></div></div>`;
      }
      const timeStr = (ep.start||'').split(' ')[1] ? `${(ep.start||'').split(' ')[1].slice(0,5)} - ${(ep.end||'').split(' ')[1]?.slice(0,5)||''}` : '';
      return `<div class="epg-entry">
        <span class="epg-tag ${i===0?'now':'next'}">${i===0?'AHORA':'SIGUIENTE'}</span>
        <span class="epg-time">${timeStr}</span>
        <div style="font-weight:600;font-size:.85rem;margin-top:.4rem;">${esc(title)}</div>
        ${desc?`<p>${esc(desc)}</p>`:''}
        ${pctHtml}
      </div>`;
    }).join('');
  }catch(e){ box.innerHTML = '<p style="color:var(--text-dim);font-size:.8rem;">No se pudo cargar la EPG.</p>'; }
}

/* ========================= VOD (movies / series) ========================= */
async function openVodTab(type){
  state.currentType = type;
  document.getElementById('vod-title').textContent = type==='movie' ? 'Películas' : 'Series';
  try{ await ensureTypeLoaded(type); }catch(e){ toast('Error cargando contenido','error'); return; }
  renderVodCategories();
  state.vodRenderCount = 60;
  applyVodFilters();
}
function renderVodCategories(){
  const type = state.currentType;
  renderCategoriesCol(document.getElementById('vod-categories-col'), type, state.currentCategory, (catId)=>{
    state.currentCategory = catId; state.vodRenderCount = 60;
    renderVodCategories();
    applyVodFilters();
    setSectionForView('view-vod','grid');
  });
}
function applyVodFilters(){
  const type = state.currentType;
  let items = state.currentCategory ? state.content[type].filter(i=>String(i.category_id)===String(state.currentCategory)) : getVisibleContent(type);
  const q = (document.getElementById('vod-search').value||'').toLowerCase();
  if(q) items = items.filter(i=>i.name.toLowerCase().includes(q));
  const sort = document.getElementById('vod-sort').value;
  items = items.slice();
  if(sort==='az') items.sort((a,b)=>a.name.localeCompare(b.name));
  else if(sort==='za') items.sort((a,b)=>b.name.localeCompare(a.name));
  else if(sort==='rating') items.sort((a,b)=>(b.rating||0)-(a.rating||0));
  else if(sort==='recent') items.sort((a,b)=>(b.added||0)-(a.added||0));
  state.vodFiltered = items;
  document.getElementById('vod-count').textContent = `${items.length} elementos`;
  renderVodGrid();
}
function renderVodGrid(){
  const grid = document.getElementById('vod-grid');
  const type = state.currentType;
  const slice = state.vodFiltered.slice(0, state.vodRenderCount);
  grid.classList.toggle('list-mode', state.viewMode.vod==='list');
  if(!slice.length){ grid.innerHTML = `<div class="empty-state" style="grid-column:1/-1;"><div class="em-icon">🔍</div><p>No se encontraron resultados.</p></div>`; document.getElementById('vod-load-more-wrap').classList.add('hidden'); return; }
  grid.innerHTML = slice.map(item=>cardTemplate(item, type)).join('');
  wireCardEvents(grid, type);
  const moreWrap = document.getElementById('vod-load-more-wrap');
  moreWrap.classList.toggle('hidden', state.vodRenderCount >= state.vodFiltered.length);
}
document.addEventListener('click', (e)=>{
  if(e.target && e.target.id==='vod-load-more'){ state.vodRenderCount += 60; renderVodGrid(); }
});

function cardTemplate(item, type, progressPct){
  const fav = isFavorite(type, item.id);
  const poster = item.icon || 'logo.png';
  const ratingHtml = item.rating ? `<span class="card-rating">⭐ ${item.rating.toFixed(1)}</span>` : '';
  const isNew = item.added && (Date.now()/1000 - item.added) < 7*24*3600;
  const progHtml = (progressPct!=null) ? `<div class="card-progress"><div class="card-progress-bar" style="width:${progressPct}%"></div></div>` : '';
  return `<div class="card" data-type="${type}" data-id="${item.id}">
    <div class="card-poster-wrap">
      <img class="card-poster" src="${esc(poster)}" loading="lazy" onerror="this.src='logo.png'">
      ${ratingHtml}
      ${isNew?'<span class="card-badge-new">NUEVO</span>':''}
      <button class="card-fav ${fav?'active':''}" data-favbtn="1">${fav?'❤️':'🤍'}</button>
      <div class="card-overlay"><span class="card-play">▶️</span></div>
      ${progHtml}
    </div>
    <div class="card-title">${esc(item.name)}</div>
  </div>`;
}
function wireCardEvents(container, forcedType){
  container.querySelectorAll('.card').forEach(card=>{
    const type = card.dataset.type || forcedType;
    const id = card.dataset.id;
    card.querySelector('[data-favbtn]').onclick = (e)=>{
      e.stopPropagation();
      const item = findItemAnySource(type, id);
      if(item){ toggleFavorite(type, item); refreshFavButtons(); }
    };
    card.onclick = ()=>{
      const item = findItemAnySource(type, id);
      if(!item) return;
      if(type==='live') playLiveFromCard(item);
      else openInfoModal(type, item);
    };
  });
}
function refreshFavButtons(){
  document.querySelectorAll('.card').forEach(card=>{
    const type = card.dataset.type, id = card.dataset.id;
    const btn = card.querySelector('[data-favbtn]');
    if(!btn) return;
    const fav = isFavorite(type,id);
    btn.classList.toggle('active', fav);
    btn.textContent = fav?'❤️':'🤍';
  });
}
function findItemAnySource(type, id){
  if(type==='live'){
    return (state.content.live||[]).find(i=>String(i.id)===String(id)) || state.favorites.find(f=>f.type==='live'&&String(f.id)===String(id));
  }
  const inType = (state.content[type]||[]).find(i=>String(i.id)===String(id));
  if(inType) return inType;
  const favItem = state.favorites.find(f=>f.type===type && String(f.id)===String(id));
  if(favItem) return favItem;
  const prog = Object.values(state.progress).find(p=>p.type===type && String(p.id)===String(id));
  return prog || null;
}
function playLiveFromCard(item){
  switchTab('live');
  setTimeout(()=>{ playLive(item); }, 150);
}

/* ========================= FAVORITES / CONTINUE (simple grid) ========================= */
function openSimpleTab(mode){
  document.getElementById('simple-title').textContent = mode==='favorites' ? '❤️ Mis Favoritos' : '🕒 Continuar Viendo';
  renderSimpleGrid(mode);
}
function renderSimpleGrid(mode){
  const grid = document.getElementById('simple-grid');
  let items = [];
  if(mode==='favorites'){
    items = state.favorites.map(f=>({...f, __progress:null}));
  } else {
    items = Object.values(state.progress).map(p=>({...p, __progress: p.duration ? (p.time/p.duration*100) : 0}))
      .sort((a,b)=>(b.lastUpdated||0)-(a.lastUpdated||0));
  }
  if(state.simpleFilter!=='all') items = items.filter(i=>i.type===state.simpleFilter);
  const q = (document.getElementById('simple-search').value||'').toLowerCase();
  if(q) items = items.filter(i=>(i.name||'').toLowerCase().includes(q));
  document.getElementById('simple-count').textContent = `${items.length} elementos`;
  grid.classList.toggle('list-mode', state.viewMode.simple==='list');
  if(!items.length){
    grid.innerHTML = `<div class="empty-state" style="grid-column:1/-1;"><div class="em-icon">${mode==='favorites'?'🤍':'🕒'}</div><p>${mode==='favorites'?'Aún no tienes favoritos.':'No hay contenido en progreso.'}</p></div>`;
    return;
  }
  grid.innerHTML = items.map(it=>cardTemplate(it, it.type, it.__progress)).join('');
  container_wireSimple(grid);
  state._simpleMode = mode;
}
function container_wireSimple(grid){
  grid.querySelectorAll('.card').forEach(card=>{
    const type = card.dataset.type, id = card.dataset.id;
    card.querySelector('[data-favbtn]').onclick = (e)=>{
      e.stopPropagation();
      const item = findItemAnySource(type, id);
      if(item){ toggleFavorite(type, item); renderSimpleGrid(state._simpleMode); }
    };
    card.onclick = ()=>{
      const item = findItemAnySource(type, id);
      if(!item){ toast('Contenido no disponible','error'); return; }
      if(type==='live') playLiveFromCard(item);
      else openInfoModal(type, item);
    };
  });
}

/* ========================= INFO MODAL ========================= */
async function openInfoModal(type, item){
  const overlay = document.getElementById('info-modal');
  const box = document.getElementById('info-modal-box');
  overlay.classList.remove('hidden');
  box.innerHTML = `<button class="modal-close" id="info-modal-close">✕</button>
    <div class="skeleton-modal"><div class="skeleton-card sk-a"></div><div class="skeleton-card sk-b"></div></div>`;
  wireModalClose();

  let plot='', cast='', director='', genre='', rating=item.rating, releaseDate='', duration='', backdrop=item.icon, poster=item.icon, episodesData=null;
  try{
    if(type==='movie'){
      const res = await xtreamFetch('get_vod_info', {vod_id:item.id});
      const d = (res && res.info) || {};
      plot = d.plot || d.description || '';
      cast = d.cast || ''; director = d.director || ''; genre = d.genre || '';
      rating = parseFloat(d.rating || d.rating_imdb || rating || 0) || null;
      releaseDate = d.releasedate || d.release_date || '';
      duration = d.duration || '';
      backdrop = (d.backdrop_path && d.backdrop_path[0]) || d.movie_image || d.cover_big || backdrop;
      poster = d.movie_image || d.cover_big || poster;
      if(res && res.movie_data && res.movie_data.container_extension) item.extension = res.movie_data.container_extension;
    } else if(type==='series'){
      const res = await xtreamFetch('get_series_info', {series_id:item.id});
      const d = (res && res.info) || {};
      plot = d.plot || ''; cast = d.cast || ''; director = d.director || ''; genre = d.genre || '';
      rating = parseFloat(d.rating || rating || 0) || null;
      releaseDate = d.releaseDate || d.releasedate || '';
      backdrop = (d.backdrop_path && d.backdrop_path[0]) || d.cover || backdrop;
      poster = d.cover || poster;
      episodesData = (res && res.episodes) || {};
    }
  }catch(e){ toast('No se pudo cargar toda la información','error'); }

  const stars = rating ? starsHtml(rating) : '';
  const metaParts = [];

  for(let i=0;i<5;i++) s += i<n ? '★' : '☆';
  return s;
}

function openCinema(meta){
  stopLiveVideo(); // asegura que el reproductor de TV en vivo no siga consumiendo recursos en segundo plano
  document.getElementById('cinema-overlay').classList.remove('hidden');
  document.getElementById('cinema-title').textContent = meta.title;
  document.getElementById('cinema-subtitle').textContent = meta.subtitle||'';
  const sidebar = document.getElementById('episodes-sidebar');
  if(meta.seriesContext){ sidebar.classList.remove('hidden'); renderEpisodesSidebar(meta.seriesContext, meta.id); }
  else sidebar.classList.add('hidden');
  const video = document.getElementById('cinema-video');
  initHlsPlayer(video, meta.url, {
    controller: state.cinemaController, url: meta.url, resumeTime: meta.resumeTime,
    progressKey: progressKey(meta.type, meta.id),
    meta: { type:meta.type, id:meta.id, name:meta.title, icon:meta.icon, extension:meta.extension, seriesId: meta.seriesContext?meta.seriesContext.item.id:null, season: meta.seriesContext?meta.seriesContext.currentSeason:null },
    onEnded: ()=>handlePlaybackEnded(meta)
  });
}
function renderEpisodesSidebar(seriesCtx, activeId){
  const eps = seriesCtx.episodesData[seriesCtx.currentSeason] || [];
  document.getElementById('episodes-list').innerHTML = eps.map(ep=>`
    <div class="episode-row ${String(ep.id)===String(activeId)?'active':''}" data-epid="${ep.id}">
      <img src="${esc((ep.info&&ep.info.movie_image)||seriesCtx.item.poster||'logo.png')}" onerror="this.src='logo.png'">
      <div><div class="ep-num">T${esc(seriesCtx.currentSeason)} · E${esc(ep.episode_num)}</div><div class="ep-title">${esc(ep.title||'')}</div></div>
    </div>`).join('');
  document.querySelectorAll('.episode-row').forEach(row=>{
    row.onclick = ()=>{
      const ep = eps.find(e=>String(e.id)===row.dataset.epid);
      playSeriesEpisode(seriesCtx, ep);
    };
  });
}
function playSeriesEpisode(seriesCtx, episode){
  const eps = seriesCtx.episodesData[seriesCtx.currentSeason] || [];
  const idx = eps.findIndex(e=>String(e.id)===String(episode.id));
  let next = eps[idx+1] || null;
  const url = buildStreamUrl('series', episode.id, episode.container_extension);
  const saved = state.progress[progressKey('series', episode.id)];
  openCinema({
    type:'series', id:episode.id,
    title: seriesCtx.item.name,
    subtitle: `Temporada ${seriesCtx.currentSeason} · Episodio ${episode.episode_num} - ${episode.title||''}`,
    url, extension: episode.container_extension,
    icon: (episode.info&&episode.info.movie_image) || seriesCtx.item.poster,
    resumeTime: saved?saved.time:0,
    seriesContext: seriesCtx,
    nextEpisode: next,
  });
}
function handlePlaybackEnded(meta){
  if(!meta.seriesContext || !meta.nextEpisode){ return; }
  const toastEl = document.getElementById('next-ep-toast');
  toastEl.classList.remove('hidden');
  let count = 5;
  document.getElementById('next-ep-count').textContent = count;
  const interval = setInterval(()=>{
    count--; document.getElementById('next-ep-count').textContent = count;
    if(count<=0){ clearInterval(interval); toastEl.classList.add('hidden'); playSeriesEpisode(meta.seriesContext, meta.nextEpisode); }
  }, 1000);
  document.getElementById('next-ep-cancel').onclick = ()=>{ clearInterval(interval); toastEl.classList.add('hidden'); };
  document.getElementById('next-ep-now').onclick = ()=>{ clearInterval(interval); toastEl.classList.add('hidden'); playSeriesEpisode(meta.seriesContext, meta.nextEpisode); };
}
function closeCinema(){
  const video = document.getElementById('cinema-video');
  try{ video.pause(); }catch(e){}
  if(video._hls){ try{video._hls.destroy();}catch(e){} video._hls=null; }
  video.removeAttribute('src');
  clearInterval(video._progressInterval);
  document.getElementById('cinema-overlay').classList.add('hidden');
}

function createController(video, root, opts={}){
  const playBtn = root.querySelector('.pc-play');
  const seek = root.querySelector('.pc-seek');
  const timeEl = root.querySelector('.pc-time');
  const muteBtn = root.querySelector('.pc-mute');
  const volume = root.querySelector('.pc-volume');
  const qualityBtn = root.querySelector('.pc-quality');
  const qualityMenu = root.querySelector('.pc-quality-menu');
  const pipBtn = root.querySelector('.pc-pip');
  const fsBtn = root.querySelector('.pc-fullscreen');
  const liveBadge = root.querySelector('.pc-live-badge');
  const container = root.closest('.player-container');

  playBtn.onclick = ()=> video.paused ? video.play().catch(()=>{}) : video.pause();
  video.addEventListener('play', ()=>playBtn.textContent='⏸️');
  video.addEventListener('pause', ()=>playBtn.textContent='▶️');
  video.addEventListener('timeupdate', ()=>{
    const isLive = seek.style.display==='none';
    if(!isLive && video.duration && isFinite(video.duration)){
      seek.value = (video.currentTime/video.duration)*100;
      timeEl.textContent = `${fmtTime(video.currentTime)} / ${fmtTime(video.duration)}`;
    }
  });
  seek.addEventListener('input', ()=>{ if(video.duration) video.currentTime = (seek.value/100)*video.duration; });
  muteBtn.onclick = ()=>{ video.muted = !video.muted; muteBtn.textContent = video.muted?'🔇':'🔊'; };
  volume.addEventListener('input', ()=>{ video.volume = parseFloat(volume.value); video.muted=false; muteBtn.textContent = volume.value==0?'🔇':'🔊'; });
  qualityBtn.onclick = ()=> qualityMenu.classList.toggle('hidden');
  pipBtn.onclick = async ()=>{
    try{ if(document.pictureInPictureElement) await document.exitPictureInPicture(); else await video.requestPictureInPicture(); }
    catch(e){ toast('PIP no disponible en este contenido','error'); }
  };
  fsBtn.onclick = ()=>{ if(!document.fullscreenElement) container.requestFullscreen?.(); else document.exitFullscreen?.(); };

  let hideTimer;
  function showControls(){ root.classList.add('visible'); clearTimeout(hideTimer); hideTimer=setTimeout(()=>{ if(!video.paused) root.classList.remove('visible'); },3000); }
  container.addEventListener('mousemove', showControls);
  container.addEventListener('click', showControls);
  showControls();

  return {
    setLiveMode(isLive){
      seek.style.display = isLive ? 'none' : '';
      timeEl.style.display = isLive ? 'none' : '';
      liveBadge.style.display = isLive ? 'inline-flex' : 'none';
    },
    setLevels(hls){
      if(!hls.levels || !hls.levels.length){ qualityMenu.innerHTML=''; return; }
      qualityMenu.innerHTML = '<div class="pc-q-item active" data-level="-1">Auto</div>' +
        hls.levels.map((l,i)=>`<div class="pc-q-item" data-level="${i}">${l.height}p</div>`).join('');
      qualityMenu.querySelectorAll('.pc-q-item').forEach(it=>{
        it.onclick = ()=>{
          hls.currentLevel = parseInt(it.dataset.level);
          qualityMenu.querySelectorAll('.pc-q-item').forEach(x=>x.classList.remove('active'));
          it.classList.add('active');
          qualityMenu.classList.add('hidden');
        };
      });
    },
    clearLevels(){ qualityMenu.innerHTML=''; }
  };
}

function initHlsPlayer(video, url, options={}){
  const wrap = video.parentElement;
  const spinner = wrap.querySelector('.pc-spinner');
  const errorBox = wrap.querySelector('.pc-error');
  errorBox.classList.add('hidden');
  spinner.classList.remove('hidden');
  if(video._hls){ try{video._hls.destroy();}catch(e){} video._hls=null; }
  clearInterval(video._progressInterval);
  video.removeAttribute('src');

  if(options.controller) options.controller.setLiveMode(!!options.isLive);

  function onReady(){
    spinner.classList.add('hidden');
    if(options.resumeTime) video.currentTime = options.resumeTime;
    video.play().catch(()=>{});
  }

  const isHlsStream = /\.m3u8($|\?)/i.test(url);

  if(isHlsStream){
    if(window.Hls && Hls.isSupported()){
      const hls = new Hls({maxBufferLength:30});
      video._hls = hls;
      hls.loadSource(url);
      hls.attachMedia(video);
      hls.on(Hls.Events.MANIFEST_PARSED, ()=>{ onReady(); options.controller && options.controller.setLevels(hls); });
      hls.on(Hls.Events.ERROR, (evt,data)=>{
        if(data && data.fatal){ spinner.classList.add('hidden'); showPlayerError(video, options); }
      });
    } else if(video.canPlayType('application/vnd.apple.mpegurl')){
      video.src = url;
      video.addEventListener('loadedmetadata', onReady, {once:true});
      video.addEventListener('error', ()=>showPlayerError(video, options), {once:true});
    } else {
      toast('Tu navegador no soporta HLS','error');
    }
  } else {
    options.controller && options.controller.clearLevels();
    video.src = url;
    video.addEventListener('loadedmetadata', onReady, {once:true});
    video.addEventListener('error', ()=>{
      spinner.classList.add('hidden');
      showPlayerError(video, options);
    }, {once:true});
  }

  video.onwaiting = ()=>spinner.classList.remove('hidden');
  video.onplaying = ()=>spinner.classList.add('hidden');

  if(options.progressKey){
    video._progressInterval = setInterval(()=>{
      if(video.duration && isFinite(video.duration) && !video.paused){
        const pct = video.currentTime/video.duration;
        if(pct < 0.95){ saveProgress(options.progressKey, {...options.meta, time:video.currentTime, duration:video.duration, lastUpdated:Date.now()}); }
        else { removeProgress(options.progressKey); }
      }
    }, 5000);
  }
  video.onended = ()=>{
    if(options.progressKey) removeProgress(options.progressKey);
    options.onEnded && options.onEnded();
  };
}
function showPlayerError(video, options){
  const wrap = video.parentElement;
  const errorBox = wrap.querySelector('.pc-error');
  errorBox.classList.remove('hidden');
  const retryBtn = errorBox.querySelector('button');
  retryBtn.onclick = ()=>{ errorBox.classList.add('hidden'); initHlsPlayer(video, options.url, options); };
}

function switchTab(tab){
  if(state.currentTab === 'live' && tab !== 'live'){ stopLiveVideo(); }

  state.currentTab = tab;
  document.querySelectorAll('.nav-btn').forEach(b=>b.classList.toggle('active', b.dataset.tab===tab));
  document.getElementById('view-live').classList.toggle('hidden', tab!=='live');
  document.getElementById('view-vod').classList.toggle('hidden', !(tab==='movies'||tab==='series'));
  document.getElementById('view-simple').classList.toggle('hidden', !(tab==='favorites'||tab==='continue'));
  document.getElementById('topbar-nav').classList.remove('open');

  state.currentCategory = null;
  if(tab==='live'){ setSectionForView('view-live','categories'); openLiveTab(); }
  else if(tab==='movies'){ setSectionForView('view-vod','categories'); openVodTab('movie'); }
  else if(tab==='series'){ setSectionForView('view-vod','categories'); openVodTab('series'); }
  else if(tab==='favorites') openSimpleTab('favorites');
  else if(tab==='continue') openSimpleTab('continue');
}

function wireKeyboard(){
  document.addEventListener('keydown', (e)=>{
    const tag = (e.target.tagName||'').toLowerCase();
    if(tag==='input' || tag==='select' || tag==='textarea') return;

    if(e.key==='Escape'){
      document.getElementById('shortcuts-modal').classList.add('hidden');
      document.getElementById('parental-modal').classList.add('hidden');
      if(!document.getElementById('info-modal').classList.contains('hidden')) closeInfoModal();
      else if(!document.getElementById('cinema-overlay').classList.contains('hidden')) closeCinema();
      return;
    }
    if(e.key==='?'){ document.getElementById('shortcuts-modal').classList.remove('hidden'); return; }

    const cinemaOpen = !document.getElementById('cinema-overlay').classList.contains('hidden');
    const video = cinemaOpen ? document.getElementById('cinema-video') : document.getElementById('live-video');
    if(!video) return;

    if(e.key===' '){ e.preventDefault(); video.paused?video.play().catch(()=>{}):video.pause(); }
    else if(e.key==='ArrowUp'){ e.preventDefault(); video.volume = Math.min(1, video.volume+0.1); }
    else if(e.key==='ArrowDown'){ e.preventDefault(); video.volume = Math.max(0, video.volume-0.1); }
    else if(e.key==='ArrowLeft'){ if(video.duration) video.currentTime = Math.max(0, video.currentTime-10); }
    else if(e.key==='ArrowRight'){ if(video.duration) video.currentTime = Math.min(video.duration, video.currentTime+10); }
    else if(e.key.toLowerCase()==='f'){ const c = video.closest('.player-container'); if(!document.fullscreenElement) c.requestFullscreen?.(); else document.exitFullscreen?.(); }
  });
}

  if(u.exp_date && u.exp_date !== '0'){
